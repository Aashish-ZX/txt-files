// push operation

void push(int data) {
    if (top == size - 1) {
        System.out.println("Stack Overflow");
    } else {
        top++;
        stack[top] = data;
    }
}

//pop operation
void pop() {
    if (top == -1) {
        System.out.println("Stack Underflow");
    } else {
        System.out.println("Deleted element: " + stack[top]);
        top--;
    }
}

//Queue enqueue
void enqueue(int data) {
    if (rear == size - 1) {
        System.out.println("Queue Overflow");
    } else {
        if (front == -1) {
            front = 0;
        }
        rear++;
        queue[rear] = data;
    }
}

// Dequeue
void dequeue() {
    if (front == -1 || front > rear) {
        System.out.println("Queue Underflow");
    } else {
        System.out.println("Deleted: " + queue[front]);
        front++;
    }
}

// singly linked list

//Node
class Node {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        next = null;
    }
}

Node head = null;

//Insert at beginning
void insertBeg(int data) {
    Node n = new Node(data);

    n.next = head;
    head = n;
}

//insert at end
void insertEnd(int data) {
    Node n = new Node(data);

    if (head == null) {
        head = n;
    } else {
        Node temp = head;

        while (temp.next != null) {
            temp = temp.next;
        }

        temp.next = n;
    }
}

//Delete from beginning
void deleteBeg() {
    if (head == null) {
        System.out.println("Empty");
    } else {
        head = head.next;
    }
}

//delete from endvoid deleteEnd() {
    if (head == null) {
        System.out.println("Empty");
    }
    else if (head.next == null) {
        head = null;
    }
    else {
        Node temp = head;

        while (temp.next.next != null) {
            temp = temp.next;
        }

        temp.next = null;
    }
}

//Display
void display() {
    Node temp = head;

    while (temp != null) {
        System.out.print(temp.data + " ");
        temp = temp.next;
    }
}

//Doubly Linked List (DLL)
//Node
class Node {
    int data;
    Node next, prev;

    Node(int data) {
        this.data = data;
    }
}

Node head = null;

//insert at beginningvoid insertBeg(int data) {
    Node n = new Node(data);

    n.next = head;

    if (head != null)
        head.prev = n;

    head = n;
}

//insert at endvoid insertEnd(int data) {
    Node n = new Node(data);

    if (head == null) {
        head = n;
    } else {
        Node temp = head;

        while (temp.next != null) {
            temp = temp.next;
        }

        temp.next = n;
        n.prev = temp;
    }
}

//Delete from beginningvoid deleteBeg() {

        return;

    head = head.next;

    if (head != null)
        head.prev = null;
}

        head = null;
    } else {
        Node temp = head;

    }
}

//Circular Linked list
//Node
class Node {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        next = null;
    }
}

Node head = null;

//Insert at beggening
void insertBeg(int data) {
    Node n = new Node(data);

    if (head == null) {
        head = n;
        n.next = head;
    } else {
        Node temp = head;

        while (temp.next != head) {
            temp = temp.next;
        }

        n.next = head;
        temp.next = n;
        head = n;
    }
}

//Insert at end
void insertEnd(int data) {
    Node n = new Node(data);

    if (head == null) {
        head = n;
        n.next = head;
    } else {
        Node temp = head;

        while (temp.next != head) {
            temp = temp.next;
        }

        temp.next = n;
        n.next = head;
    }
}

//Delete form beginning
void deleteBeg() {
    if (head == null) {
        System.out.println("Empty");
    }

    else if (head.next == head) {
        head = null;
    }

    else {
        Node temp = head;

        while (temp.next != head) {
            temp = temp.next;
        }

        temp.next = head.next;
        head = head.next;
    }
}

//Delete from end
void deleteEnd() {
    if (head == null) {
        System.out.println("Empty");
    }

    else if (head.next == head) {
        head = null;
    }

    else {
        Node temp = head;

        while (temp.next.next != head) {
            temp = temp.next;
        }

        temp.next = head;
    }
}

//Display 
void display() {

    if (head == null) {
        System.out.println("Empty");
        return;
    }

    Node temp = head;

    do {
        System.out.print(temp.data + " ");
        temp = temp.next;
    }

    while (temp != head);
}

//Binary Tree Insertion
Node insert(Node root, int data) {

    if (root == null) {
        root = new Node(data);
        return root;
    }

    if (data < root.data) {
        root.left = insert(root.left, data);
    }

    else {
        root.right = insert(root.right, data);
    }

    return root;
}

//Deletion BST
Node min(Node root) {

    while(root.left != null) {
        root = root.left;
    }

    return root;
}

//Tree Traversal
//inorder
void inorder(Node root) {

    if (root != null) {

        inorder(root.left);

        System.out.print(root.data+" ");

        inorder(root.right);
    }
}

//Pre order
void preorder(Node root) {

    if (root != null) {

        System.out.print(root.data+" ");

        preorder(root.left);

        preorder(root.right);
    }
}

//post order
void postorder(Node root) {

    if (root != null) {

        postorder(root.left);

        postorder(root.right);

        System.out.print(root.data+" ");
    }
}
