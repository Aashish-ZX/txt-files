#include <stdio.h>
#include <stdlib.h>

int main()
{
    int queue[] = {12, 15, 18, 9, 7, 59, 89, 17};
    int head = 5;
    int total = 0;

    for(int i = 0; i < 8; i++)
    {
        total = total + abs(queue[i] - head);
        head = queue[i];
    }

    printf("Total head movement = %d", total);

    return 0;
}


#include <stdio.h>
#include <stdlib.h>

int main()
{
    int queue[] = {12, 15, 18, 9, 7, 59, 89, 17};
    int visited[8] = {0};
    int head = 5;
    int total = 0;

    for(int count = 0; count < 8; count++)
    {
        int min = 9999;
        int pos = -1;

        for(int i = 0; i < 8; i++)
        {
            if(visited[i] == 0)
            {
                int distance = abs(queue[i] - head);

                if(distance < min)
                {
                    min = distance;
                    pos = i;
                }
            }
        }

        visited[pos] = 1;
        total = total + min;
        head = queue[pos];
    }

    printf("Total head movement = %d", total);

    return 0;
}
